\BookSection{Α. Βασικές αρχές}
\begin{itemize}
\item \textbf{Κοινό ύφος:} λευκό φόντο, καθαρές γραμμές, χωρίς σκιά ή περιττό τρισδιάστατο εφέ.
\item \textbf{Αναγνώσιμα βέλη:} σταθερό πάχος και κοινή αιχμή για διανύσματα ίδιου τύπου.
\item \textbf{Φυσική σημασία:} οι φοράς των αξόνων και οι θετικές κατευθύνσεις γράφονται ρητά.
\item \textbf{Δύο μελάνια:} οι διαφορές φαίνονται και με συνεχείς/διακεκομμένες γραμμές, όχι μόνο από το χρώμα.
\item \textbf{Μονάδες:} πάντα όρθιες, π.χ. $x\,(\mathrm{m})$ και $t\,(\mathrm{s})$.
\end{itemize}

\BookSection{Ζ. Παραπομπές, στυλ και γρήγορος έλεγχος}
\noindent Ο κώδικας κάθε διαγράμματος μπαίνει στο αρχείο κεφαλαίου, αλλά \textbf{οι κανόνες εμφάνισης} μένουν στο \texttt{styles/book-figures.tex}.
\begin{itemize}
 \item Άξονας: \texttt{book/axis} · ταχύτητα: \texttt{book/velocity}.\par Επιτάχυνση: \texttt{book/accel}.
 \item Βάρος: \texttt{book/weight} · δύναμη: \texttt{book/force} · τριβή: \texttt{book/friction}.
 \item Βοηθητική γραμμή: \texttt{book/guide}.\par Καμπύλες PGFPlots: \texttt{book/seriesA} και \texttt{book/seriesB}.
 \item Γράφημα PGFPlots: \texttt{\textbackslash begin\{axis\}[book/plot,...]}.
 \item Κάτω από \textit{κάθε} σχήμα: \texttt{\textbackslash caption\{...\}} και \texttt{\textbackslash label\{fig:...\}}.
 \item Μην χρησιμοποιείς \texttt{red}, \texttt{blue}, \texttt{orange}, \texttt{black!40} απευθείας μέσα στα σχήματα.
\end{itemize}
\noindent\textbf{Πριν από την παράδοση:} έλεγξε γράμματα/μονάδες, φορά θετικών αξόνων, διανύσματα, αν τα διακεκομμένα διακρίνονται στην ασπρόμαυρη εκτύπωση και αν η εκτυπωτική μορφή έχει μόνο δύο πλάκες.
