% placement; these paths do not set or enlarge the material header's height.
\newcommand{\BookWorksheetHeaderIcon}{%
  \begin{tikzpicture}[x=\BookMaterialHeaderIconSize,y=\BookMaterialHeaderIconSize,
    line width=\BookMaterialHeaderIconLineWidth,line cap=round,line join=round]
    \path[use as bounding box] (0,0) rectangle (1,1);
    \draw[white] (.03,.03)--(.12,.29)--(.77,.94)
      .. controls (.81,.98) and (.85,.98) .. (.89,.94)
      --(.94,.89).. controls (.98,.85) and (.98,.81) .. (.94,.77)
      --(.29,.12)--cycle;
    \draw[white] (.12,.29)--(.29,.12);
    \draw[white] (.77,.94)--(.94,.77);

  \end{tikzpicture}%
