\section{Πυκνά μαθηματικά με σαφείς αναφορές}
\texttt{DENSEMATH}\par
\begin{align}
 f(x)&=\alpha x+\beta,\label{eq:stress-function}\\
 f(x+h)-f(x)&=\alpha h,\\
 \lim_{h\to0}\frac{f(x+h)-f(x)}{h}&=\alpha,\\
 \int_a^b(\alpha x+\beta)\,\mathrm dx
 &=\frac{\alpha}{2}(b^2-a^2)+\beta(b-a).
\end{align}
Η σχέση~\eqref{eq:stress-function} συνδέει τα παραδείγματα του κεφαλαίου.
\begin{definitionbox}[Γραμμική συνάρτηση]
Για $\alpha\ne0$, η γραφική παράσταση της $f$ είναι ευθεία.
\end{definitionbox}
\begin{theorembox}[Μεταβολή συνάρτησης]
Για κάθε $x,h\in\mathbb R$, ισχύει $f(x+h)-f(x)=\alpha h$.
\end{theorembox}
\begin{proof}Αντικαθιστούμε τον τύπο της συνάρτησης και απλοποιούμε.\end{proof}
